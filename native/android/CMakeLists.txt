cmake_minimum_required(VERSION 3.22)
project(QuestPDFAndroid LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK CMake toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_link_options(-Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
    -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,--no-undefined)

if(QUESTPDF_BUILD_QPDF)
    set(BUILD_STATIC_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(USE_IMPLICIT_CRYPTO OFF CACHE BOOL "" FORCE)
    set(REQUIRE_CRYPTO_NATIVE ON CACHE BOOL "" FORCE)
    add_subdirectory("${QPDF_DIR}" qpdf EXCLUDE_FROM_ALL)
    # ABI 4 predates the upstream filename change. Set filename AND SONAME.
    set_target_properties(libqpdf PROPERTIES OUTPUT_NAME questpdf_qpdf VERSION "" SOVERSION "")
endif()

if(QUESTPDF_BUILD_SKIA)
    file(GLOB_RECURSE WRAPPER_SOURCES CONFIGURE_DEPENDS "${WRAPPER_DIR}/src/*.cpp")
    add_library(questpdf_skia SHARED ${WRAPPER_SOURCES})
    target_include_directories(questpdf_skia PRIVATE "${SKIA_DIR}")
    target_compile_options(questpdf_skia PRIVATE -fno-rtti -ffunction-sections -fdata-sections)
    file(GLOB SKIA_ARCHIVES "${SKIA_OUTPUT_DIR}/*.a")
    if(NOT SKIA_ARCHIVES)
        message(FATAL_ERROR "Build Skia archives before configuring the wrapper")
    endif()
    target_link_libraries(questpdf_skia PRIVATE -Wl,--start-group ${SKIA_ARCHIVES}
        -Wl,--end-group android log z m dl)
endif()
